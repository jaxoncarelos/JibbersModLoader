cmake_minimum_required(VERSION 3.18)
project(JibbersMacHooks C ASM CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
add_subdirectory(${DOBBY_SOURCE} dobby)
add_library(jibbers_hooks SHARED hooks.c dispatch.S)
target_include_directories(jibbers_hooks PRIVATE ${DOBBY_SOURCE}/include)
target_link_libraries(jibbers_hooks PRIVATE dobby_static)
set_target_properties(jibbers_hooks PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
add_executable(hook_smoke smoke.c)
target_link_libraries(hook_smoke PRIVATE jibbers_hooks)
